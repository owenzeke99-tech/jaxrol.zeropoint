local SAVE_FILE_NAME = "ZeroPoint_Session.json"
local EXPIRE_TIME = 24 * 60 * 60 -- 24 hours in seconds

-- Check if user already verified within 24 hours
local function CheckSavedKey()
    if writefile and readfile and isfile and isfile(SAVE_FILE_NAME) then
        local success, data = pcall(function()
            local HttpService = game:GetService("HttpService")
            return HttpService:JSONDecode(readfile(SAVE_FILE_NAME))
        end)
        if success and data and data.timestamp and data.key == CORRECT_KEY then
            if os.time() - data.timestamp < EXPIRE_TIME then
                return true
            end
        end
    end
    return false
end

-- Save key session silently
local function SaveKeySession()
    if writefile then
        pcall(function()
            local HttpService = game:GetService("HttpService")
            local data = {
                key = CORRECT_KEY,
                timestamp = os.time()
            }
            writefile(SAVE_FILE_NAME, HttpService:JSONEncode(data))
        end)
    end
end

-- If session is valid, load script directly without showing UI

    local success, err = pcall(function()

    end)
    if not success then
        warn("Error loading Zero Point script: " + tostring(err))
    end
    return
end

-- Smooth Horizontal Shake for Menu Error
local isShaking = false
local function ShakeError()
    if isShaking then return end
    isShaking = true
    
    local basePos = UDim2.new(0.5, 0, 0.5, 0)
    local offsets = {-12, 12, -8, 8, -4, 4, 0}
    for _, offsetX in ipairs(offsets) do
        local tween = TweenService:Create(
            MainFrame,
            TweenInfo.new(0.04, Enum.EasingStyle.Sine, Enum.EasingDirection.InOut),
            {Position = UDim2.new(0.5, offsetX, 0.5, 0)}
        )
        tween:Play()
        tween.Completed:Wait()
    end
    MainFrame.Position = basePos
    isShaking = false
end

-- Hover Effects
local function AddButtonHover(button, normalColor, hoverColor)
    button.MouseEnter:Connect(function()
        TweenService:Create(button, TweenInfo.new(0.2, Enum.EasingStyle.Quad), {BackgroundColor3 = hoverColor}):Play()
    end)
    button.MouseLeave:Connect(function()
        TweenService:Create(button, TweenInfo.new(0.2, Enum.EasingStyle.Quad), {BackgroundColor3 = normalColor}):Play()
    end)
end
